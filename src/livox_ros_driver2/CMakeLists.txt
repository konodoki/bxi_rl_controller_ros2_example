cmake_minimum_required(VERSION 3.14)
project(livox_ros_driver2)

# Default to C99
if(NOT CMAKE_C_STANDARD)
  set(CMAKE_C_STANDARD 99)
endif()

# Default to C++14
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 14)
endif()

list(INSERT CMAKE_MODULE_PATH 0 "${PROJECT_SOURCE_DIR}/cmake/modules")

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)
endif()

# Printf version info
include(cmake/version.cmake)
project(${PROJECT_NAME} VERSION ${LIVOX_ROS_DRIVER2_VERSION} LANGUAGES CXX)
message(STATUS "${PROJECT_NAME} version: ${LIVOX_ROS_DRIVER2_VERSION}")

# ROS 2 dependencies
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)

# generate custom msg headers
set(LIVOX_INTERFACES livox_interfaces2)
rosidl_generate_interfaces(${LIVOX_INTERFACES}
  "msg/CustomPoint.msg"
  "msg/CustomMsg.msg"
  DEPENDENCIES std_msgs
  LIBRARY_NAME ${PROJECT_NAME}
)

# Livox-SDK2 headers are architecture-independent. Prebuilt libraries live in
# Livox-SDK2/lib/<arch> so a workspace containing multiple target platforms
# never links a library for the wrong CPU by accident.
set(LIVOX_LIDAR_SDK_ROOT "${PROJECT_SOURCE_DIR}/Livox-SDK2" CACHE PATH
  "Path to the bundled Livox-SDK2")
set(LIVOX_SDK_ARCH "" CACHE STRING
  "Livox-SDK2 library architecture (auto-detected when empty)")

if(NOT LIVOX_SDK_ARCH)
  string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" LIVOX_SYSTEM_PROCESSOR)
  if(LIVOX_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    set(LIVOX_SDK_ARCH "x86_64")
  elseif(LIVOX_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(LIVOX_SDK_ARCH "aarch64")
  elseif(LIVOX_SYSTEM_PROCESSOR MATCHES "^(armv7|armv7l|armhf)$")
    set(LIVOX_SDK_ARCH "armv7l")
  else()
    set(LIVOX_SDK_ARCH "${LIVOX_SYSTEM_PROCESSOR}")
  endif()
endif()

set(LIVOX_LIDAR_SDK_INCLUDE_DIR "${LIVOX_LIDAR_SDK_ROOT}/include")
set(LIVOX_LIDAR_SDK_LIBRARY_DIR
  "${LIVOX_LIDAR_SDK_ROOT}/lib/${LIVOX_SDK_ARCH}")
set(LIVOX_LIDAR_SDK_STATIC_LIBRARY
  "${LIVOX_LIDAR_SDK_LIBRARY_DIR}/liblivox_lidar_sdk_static.a")
set(LIVOX_LIDAR_SDK_SHARED_LIBRARY
  "${LIVOX_LIDAR_SDK_LIBRARY_DIR}/liblivox_lidar_sdk_shared.so")

if(EXISTS "${LIVOX_LIDAR_SDK_STATIC_LIBRARY}")
  set(LIVOX_LIDAR_SDK_LIBRARY "${LIVOX_LIDAR_SDK_STATIC_LIBRARY}")
elseif(EXISTS "${LIVOX_LIDAR_SDK_SHARED_LIBRARY}")
  set(LIVOX_LIDAR_SDK_LIBRARY "${LIVOX_LIDAR_SDK_SHARED_LIBRARY}")
else()
  message(FATAL_ERROR
    "Livox-SDK2 library for '${LIVOX_SDK_ARCH}' was not found. "
    "Copy liblivox_lidar_sdk_static.a or liblivox_lidar_sdk_shared.so to "
    "${LIVOX_LIDAR_SDK_LIBRARY_DIR}. "
    "Detected CMAKE_SYSTEM_PROCESSOR='${CMAKE_SYSTEM_PROCESSOR}'.")
endif()

message(STATUS
  "Livox-SDK2 architecture: ${LIVOX_SDK_ARCH}; library: ${LIVOX_LIDAR_SDK_LIBRARY}")

# livox ros2 driver target
ament_auto_add_library(${PROJECT_NAME} SHARED
  src/livox_ros_driver2.cpp
  src/lddc.cpp
  src/driver_node.cpp
  src/lds.cpp
  src/lds_lidar.cpp

  src/comm/comm.cpp
  src/comm/ldq.cpp
  src/comm/semaphore.cpp
  src/comm/lidar_imu_data_queue.cpp
  src/comm/cache_index.cpp
  src/comm/pub_handler.cpp

  src/parse_cfg_file/parse_cfg_file.cpp
  src/parse_cfg_file/parse_livox_lidar_cfg.cpp

  src/call_back/lidar_common_callback.cpp
  src/call_back/livox_lidar_callback.cpp
)

# get include directories of custom msg headers
rosidl_get_typesupport_target(cpp_typesupport_target
  ${LIVOX_INTERFACES} "rosidl_typesupport_cpp")
target_link_libraries(${PROJECT_NAME} "${cpp_typesupport_target}")

# include directories
target_include_directories(${PROJECT_NAME} PUBLIC
  ${LIVOX_LIDAR_SDK_INCLUDE_DIR}
  3rdparty
  src
)

# link libraries
target_link_libraries(${PROJECT_NAME}
  ${LIVOX_LIDAR_SDK_LIBRARY}
)

# When an architecture provides only the shared SDK, install it beside the ROS
# libraries so the environment generated by colcon can resolve it at runtime.
if("${LIVOX_LIDAR_SDK_LIBRARY}" STREQUAL "${LIVOX_LIDAR_SDK_SHARED_LIBRARY}")
  install(FILES "${LIVOX_LIDAR_SDK_SHARED_LIBRARY}" DESTINATION lib)
endif()

rclcpp_components_register_node(${PROJECT_NAME}
  PLUGIN "livox_ros::DriverNode"
  EXECUTABLE ${PROJECT_NAME}_node
)

  # the following line skips the linter which checks for copyrights

  # the following line skips cpplint (only works in a git repo)
  # uncomment the line when this package is not in a git repo
  #set(ament_cmake_cpplint_FOUND TRUE)
  ament_lint_auto_find_test_dependencies()
endif()

ament_auto_package(INSTALL_TO_SHARE
  config
  launch
)
